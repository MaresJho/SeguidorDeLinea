const int STBY = 6;

// Motor 1 (canal A)
const int AIN1 = 7;
const int AIN2 = 8;
const int PWMA = 9;

// Motor 2 (canal B)
const int BIN1 = 11;
const int BIN2 = 12;
const int PWMB = 10;

const int VELOCIDAD = 120; // baja, para no exigir mucho al USB
const int TIEMPO_MOVIMIENTO = 2000;
const int TIEMPO_ESPERA = 5000;

void setup() {
  Serial.begin(9600);

  pinMode(STBY, OUTPUT);
  pinMode(AIN1, OUTPUT);
  pinMode(AIN2, OUTPUT);
  pinMode(PWMA, OUTPUT);
  pinMode(BIN1, OUTPUT);
  pinMode(BIN2, OUTPUT);
  pinMode(PWMB, OUTPUT);

  digitalWrite(STBY, HIGH);
}

void motor1(int velocidad) {
  digitalWrite(AIN1, HIGH);
  digitalWrite(AIN2, LOW);
  analogWrite(PWMA, velocidad);
}

void motor2(int velocidad) {
  digitalWrite(BIN1, HIGH);
  digitalWrite(BIN2, LOW);
  analogWrite(PWMB, velocidad);
}

void detenerTodo() {
  analogWrite(PWMA, 0);
  analogWrite(PWMB, 0);
}

void loop() {
  Serial.println("Moviendo motor 1");
  motor1(VELOCIDAD);
  delay(TIEMPO_MOVIMIENTO);
  detenerTodo();

  Serial.println("Esperando 5 segundos");
  delay(TIEMPO_ESPERA);

  Serial.println("Moviendo motor 2");
  motor2(VELOCIDAD);
  delay(TIEMPO_MOVIMIENTO);
  detenerTodo();

  Serial.println("Esperando 5 segundos");
  delay(TIEMPO_ESPERA);
}
